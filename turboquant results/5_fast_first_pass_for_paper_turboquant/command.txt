5. Fast first pass for paper TurboQuant

Tests:
- baseline and turbo hosts
- baseline runs f16 and marks unsupported tq rows explicitly
- quick memory and throughput signal
- one prefill-heavy cell and one decode-growth cell
- extra-lite matrix for about 10 to 20 minutes on 2x Tesla M40

docker compose -f /srv/share/ollama-benchmark/ollama-main-turboquant/docker-compose.ollama-bench.yml exec ollama-bench kvstress-bench \
  --hosts http://127.0.0.1:11438,http://127.0.0.1:11439 \
  --host-labels baseline,turbo \
  --host-kv-support legacy,request \
  --profile quick \
  --model qwen3-30b:instruct2507-udq4kxl \
  --kv-modes f16,tq35 \
  --workloads prefill-heavy,decode-growth \
  --num-ctx 8192,16384 \
  --concurrency 1 \
  --warmup 0 \
  --epochs 1 \
  --max-tokens 256 \
  --timeout 0 \
  --output /results_full/5_fast_first_pass_for_paper_turboquant/impact_paper.csv \
  --jsonl-output /results_full/5_fast_first_pass_for_paper_turboquant/impact_paper.jsonl \
